Queens Hazard Ranking

Hazard Categories

The Queens Resource Map will display ranked hazard-related preparedness and recovery resources and messaging at the Community District Tabulation Area (CDTA) level.

Our hazard categorization is based on New York City’s 2024 Jurisdictional Risk Assessment1 report. The 2024 report uses the Pennsylvania Public Health Risk Assessment Tool (PHRAT) model in order to develop a “Planning Priority Score” used to suggest where New York City should focus its resources and planning efforts. The ten categories defined by the 2024 JRA report and their rank are:

In the absence of available information at the Community District level, extreme cold, mass casualty events / active shooter incidents, and blackout & cyberattack cannot be ranked.

In order to identify a suitable ranking methodology for the remaining hazards, a discussion of the disparate data is warranted.

Code 
# Fetch data sources

cdta <- read_sf("https://services5.arcgis.com/GfwWNkhOj9bNBqoJ/arcgis/rest/services/NYC_Community_District_Tabulation_Areas_2020/FeatureServer/0/query?where=CDTAType=0&outFields=*&outSR=4326&f=pgeojson") |> 
  mutate(cdtaCode = paste0(
    BoroCode,
    str_sub(CDTA2020, -2)
  ))

stormwater_limited <- read_sf("https://github.com/BetaNYC/q-map/releases/download/data-v4/stormwater_limited_1_77.geojson") |> st_make_valid()

stormwater_moderate <- read_sf("https://github.com/BetaNYC/q-map/releases/download/data-v4/stormwater_moderate_2_13.geojson") |> 
  st_make_valid()

ss_cur_fvi <- soc_read(
  "https://data.cityofnewyork.us/Environment/New-York-City-s-Flood-Vulnerability-Index/mrjc-v9pm.json",
  query = soc_query(
    select = "the_geom, GEOID, ss_cur"
  ),
  include_synthetic_cols = FALSE
)

hvi <- read_sf("https://services1.arcgis.com/8cuieNI8NbqQZQVJ/arcgis/rest/services/HVI_by_CDTA_CRAD_2024/FeatureServer/0/query?where=1=1&outFields=*&outSR=4326&f=pgeojson")

chem_small_biz <- read_sf("https://services3.arcgis.com/A6Zjpzrub8ESZ3c7/arcgis/rest/services/Chemically%20Intensive%20Small%20Businesses_2/FeatureServer/0/query?where=1=1&outFields=*&outSR=4326&f=pgeojson")

hazmat_storage <- read_sf("https://services6.arcgis.com/yG5s3afENB5iO9fj/arcgis/rest/services/EPA_RCA_Fuel_Oil_Storage/FeatureServer/0/query?where=1=1&outFields=*&outSR=4326&f=pgeojson")

pivi <- read_sf("https://services1.arcgis.com/8cuieNI8NbqQZQVJ/arcgis/rest/services/Pandemic_Influenza_Vulnerability_Index/FeatureServer/0/query?where=1=1&outFields=*&outSR=4326&f=pgeojson")

# Aggregate where needed

## Stormwater -> CDTA
stormwater_all <- st_union(
  st_union(stormwater_limited),
  st_union(stormwater_moderate)
) |> 
  st_transform(4326) |> 
  st_make_valid()

stormwater_pct <- cdta |> 
  mutate(cdta_area = st_area(geometry)) |> 
  st_intersection(stormwater_all) |> 
  transmute(CDTA2020,
            stormwater_pct = as.numeric(st_area(geometry) / cdta_area) * 100) |>
  st_drop_geometry()

cdta <- cdta |> 
  left_join(stormwater_pct) |> 
  mutate(stormwater_pct = replace_na(stormwater_pct, 0))

## Tract-level FVI -> CDTA
# Assign max FVI to CD
ss_cur_fvi_cd <- ss_cur_fvi |> 
  st_join(cdta |> select(CDTA2020),
          largest = TRUE)

ss_cur_fvi_cd <- ss_cur_fvi_cd |> 
  st_drop_geometry() |> 
  group_by(CDTA2020) |> 
  summarize(max_ss_cur_fvi = max(ss_cur, na.rm = T))
 
cdta <- cdta |> 
  left_join(ss_cur_fvi_cd)

# Hazmat storage
cdta$hazmat_count <- lengths(st_intersects(cdta, hazmat_storage))

# Join remainder

## HVI
cdta <- cdta |> 
  left_join(hvi |> 
              st_drop_geometry() |> 
              select(CommDist, HVI),
            by = join_by(cdtaCode == CommDist))

## Chemically Intensive Small Business count
cdta <- cdta |> 
  left_join(chem_small_biz |> 
              st_drop_geometry() |> 
              select(BoroCD, top_3) |> 
              mutate(BoroCD = as.character(BoroCD)) |> 
              rename(chem_count = "top_3"),
            by = join_by(cdtaCode == BoroCD))

## PIVI
cdta <- cdta |> 
  left_join(pivi |> 
              st_drop_geometry() |> 
              select(COMMDIST, PIVI),
            by = join_by(cdtaCode == COMMDIST))

cdta <- cdta |> 
  select(BoroCode,CDTA2020, cdtaCode, CDTAName,
         stormwater_pct, max_ss_cur_fvi, hazmat_count, HVI, chem_count, PIVI) |> 
  relocate(geometry, .after = last_col()) |> 
  mutate(max_ss_cur_fvi = as.integer(max_ss_cur_fvi))

Heavy Rain

The impact of heavy rain on a given Community District is taken as the percent area that experiences stormwater flooding of any kind (i.e. greater than 4 inches of flooding without sea level rise under a limited- or moderate-intensity rainfall event). Citywide, this percentage ranges from 0.3% to 3.52%. In Queens, the percentage ranges from 0.41% to 3.08%. All 59 Community Districts experience some stormwater flooding.

Computing percentile rank in this case is straightforward:

Code 
cdta <- cdta |> 
  mutate(stormwater_pctrank = 100 * ecdf(cdta$stormwater_pct)(cdta$stormwater_pct))

cil_table(cdta |> 
            st_drop_geometry() |> 
            filter(BoroCode == 4) |>
            select(CDTAName, stormwater_pctrank) |>
            mutate(stormwater_pctrank = round(stormwater_pctrank, 2)) |> 
            rename(
              "CDTA Name" = "CDTAName",
              "Stormwater Extent Percentile (Citywide)" = "stormwater_pctrank"
            ),
          full_height = TRUE)
